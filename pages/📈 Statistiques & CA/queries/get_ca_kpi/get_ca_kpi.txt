WITH params AS (
    SELECT 
        CASE 
            WHEN EXTRACT(MONTH FROM CURRENT_DATE) >= 11 
            THEN MAKE_DATE(EXTRACT(YEAR FROM CURRENT_DATE)::int, 11, 1)
            ELSE MAKE_DATE((EXTRACT(YEAR FROM CURRENT_DATE)::int - 1), 11, 1)
        END AS debut_exercice_n,
        CASE 
            WHEN EXTRACT(MONTH FROM CURRENT_DATE) >= 11 
            THEN MAKE_DATE((EXTRACT(YEAR FROM CURRENT_DATE)::int - 1), 11, 1)
            ELSE MAKE_DATE((EXTRACT(YEAR FROM CURRENT_DATE)::int - 2), 11, 1)
        END AS debut_exercice_n1
)
SELECT 
    -- CA du Jour
    ROUND(COALESCE(SUM(bl."TOTAL_HT") FILTER (
        WHERE bl."Date_Facture" = CURRENT_DATE
    ), 0), 2) AS ca_jour,
    
    -- CA du Mois en cours
    ROUND(COALESCE(SUM(bl."TOTAL_HT") FILTER (
        WHERE DATE_TRUNC('month', bl."Date_Facture") = DATE_TRUNC('month', CURRENT_DATE)
    ), 0), 2) AS ca_mois,
    
    -- CA Exercice en cours (N)
    ROUND(COALESCE(SUM(bl."TOTAL_HT") FILTER (
        WHERE bl."Date_Facture" >= p.debut_exercice_n 
          AND bl."Date_Facture" < p.debut_exercice_n + INTERVAL '1 year'
    ), 0), 2) AS ca_exercice_actuel,

    -- CA Exercice précédent complet (N-1)
    ROUND(COALESCE(SUM(bl."TOTAL_HT") FILTER (
        WHERE bl."Date_Facture" >= p.debut_exercice_n1 
          AND bl."Date_Facture" < p.debut_exercice_n
    ), 0), 2) AS ca_exercice_precedent

FROM public."BONS_DE_LIVRAISON" bl, params p
WHERE UPPER(TRIM(COALESCE(bl."Statut_Facturation", ''))) IN ('FACTURÉ', 'FACTURE', 'PAYÉ', 'PAYE')
  AND COALESCE(bl."Origine", '') <> 'Garantie';
