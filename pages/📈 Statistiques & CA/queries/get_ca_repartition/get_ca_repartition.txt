WITH params AS (
    SELECT CASE 
        WHEN EXTRACT(MONTH FROM CURRENT_DATE) >= 11 
        THEN MAKE_DATE(EXTRACT(YEAR FROM CURRENT_DATE)::int, 11, 1)
        ELSE MAKE_DATE((EXTRACT(YEAR FROM CURRENT_DATE)::int - 1), 11, 1)
    END AS debut_exercice
),
lignes_calculees AS (
    SELECT 
        CASE 
            -- 1. L'Occasion en priorité
            WHEN UPPER(lbl."Descriptif") LIKE '%OCCASION%' THEN 'Matériel d''Occasion'
            
            -- 2. Les Robots
            WHEN UPPER(lbl."Descriptif") LIKE '%ROBOT %' 
              OR UPPER(lbl."Descriptif") LIKE '%SEGWAY%' 
              OR UPPER(lbl."Descriptif") LIKE '%AMBROGIO%'
              OR UPPER(lbl."Descriptif") LIKE '%TECHLINE%' THEN 'Robots Tondeuses'
              
            -- 3. Le Matériel à Batterie (EGO, etc.)
            WHEN UPPER(lbl."Descriptif") LIKE '%BATTERIE%' 
              OR UPPER(lbl."Descriptif") LIKE '%EGO %' THEN 'Matériel à Batterie'
              
            -- 4. Le Matériel Neuf (Thermique & Autre)
            WHEN UPPER(lbl."Descriptif") LIKE '%TONDEUSE %' 
              OR UPPER(lbl."Descriptif") LIKE '%AUTOPORTÉE%' 
              OR UPPER(lbl."Descriptif") LIKE '%AUTOPORTEE%' 
              OR UPPER(lbl."Descriptif") LIKE '%TRACTEUR %'
              OR UPPER(lbl."Descriptif") LIKE '%DEBROUSSAILLEUSE%' 
              OR UPPER(lbl."Descriptif") LIKE '%TRONCONNEUSE%' 
              OR UPPER(lbl."Descriptif") LIKE '%MOTOCULTEUR%' THEN 'Matériel Neuf (Thermique)'
              
            -- 5. L'Atelier et la Main d'œuvre
            WHEN UPPER(lbl."Descriptif") LIKE '%MAIN%OEUVRE%' 
              OR UPPER(lbl."Descriptif") LIKE '%FORFAIT%' 
              OR UPPER(TRIM(COALESCE(bl."Origine", ''))) IN ('INTERVENTION', 'LOCATION_MAINTENANCE') THEN 'Atelier & Services'
              
            -- 6. La location de flotte
            WHEN UPPER(lbl."Descriptif") LIKE '%LOCATION %' 
              OR UPPER(TRIM(COALESCE(bl."Origine", ''))) = 'LOCATION' THEN 'Location'
              
            -- 7. Tout le reste (Pièces détachées, Huile, Fil, Accessoires)
            ELSE 'Comptoir (Pièces & Accessoires)'
        END AS pole,
        
        -- Calcul du HT de la ligne avec prise en compte des remises
        ROUND((COALESCE(lbl."Quantite", 0)::numeric * COALESCE(lbl."Prix_HT", 0)::numeric) - 
              CASE 
                WHEN lbl."Type_Remise" = '%' THEN (COALESCE(lbl."Quantite", 0)::numeric * COALESCE(lbl."Prix_HT", 0)::numeric) * (COALESCE(lbl."Valeur_Remise", 0)::numeric / 100.0) 
                WHEN lbl."Type_Remise" = '€' THEN COALESCE(lbl."Valeur_Remise", 0)::numeric 
                ELSE 0 
              END, 2) AS total_ligne_ht
    FROM public."Ligne_BL" lbl
    JOIN public."BONS_DE_LIVRAISON" bl ON bl."ID_BL" = lbl."ID_BL"
    CROSS JOIN params p
    WHERE UPPER(TRIM(COALESCE(bl."Statut_Facturation", ''))) IN ('FACTURÉ', 'FACTURE', 'PAYÉ', 'PAYE')
      AND bl."Date_Facture" >= p.debut_exercice
      AND bl."Date_Facture" < p.debut_exercice + INTERVAL '1 year'
      AND COALESCE(bl."Origine", '') <> 'Garantie'
)
SELECT 
    pole AS x,
    SUM(total_ligne_ht) AS y
FROM lignes_calculees
GROUP BY pole
ORDER BY y DESC;
