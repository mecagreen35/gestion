WITH params AS (
    SELECT DATE_TRUNC('month', CURRENT_DATE)::date AS debut_mois_courant
),
mois_glissants AS (
    SELECT 
        i,
        (p.debut_mois_courant - (i * INTERVAL '1 month'))::date AS date_debut_n,
        (p.debut_mois_courant - ((i - 1) * INTERVAL '1 month'))::date AS date_fin_n,
        ((p.debut_mois_courant - (i * INTERVAL '1 month')) - INTERVAL '1 year')::date AS date_debut_n1,
        ((p.debut_mois_courant - ((i - 1) * INTERVAL '1 month')) - INTERVAL '1 year')::date AS date_fin_n1
    FROM params p, generate_series(11, 0, -1) AS i
),
factures AS (
    SELECT 
        bl."TOTAL_HT",
        bl."Date_Facture"
    FROM public."BONS_DE_LIVRAISON" bl
    WHERE bl."Date_Facture" IS NOT NULL
      AND UPPER(TRIM(COALESCE(bl."Statut_Facturation", ''))) IN ('FACTURÉ', 'FACTURE', 'PAYÉ', 'PAYE')
      AND UPPER(TRIM(COALESCE(bl."Origine", ''))) NOT IN ('GARANTIE', 'LOCATION_MAINTENANCE', 'INTERNE', 'MAINTENANCE')
)
SELECT 
    -- Libellé avec marqueur direct sur le mois actuel
    CASE 
        WHEN m.i = 0 THEN 
            CASE EXTRACT(MONTH FROM m.date_debut_n)
                WHEN 1 THEN 'Jan' WHEN 2 THEN 'Fév' WHEN 3 THEN 'Mar'
                WHEN 4 THEN 'Avr' WHEN 5 THEN 'Mai' WHEN 6 THEN 'Juin'
                WHEN 7 THEN 'Juil' WHEN 8 THEN 'Août' WHEN 9 THEN 'Sept'
                WHEN 10 THEN 'Oct' WHEN 11 THEN 'Nov' WHEN 12 THEN 'Déc'
            END || ' ' || SUBSTRING(EXTRACT(YEAR FROM m.date_debut_n)::text FROM 3 FOR 2) || ' (En cours)'
        ELSE 
            CASE EXTRACT(MONTH FROM m.date_debut_n)
                WHEN 1 THEN 'Jan' WHEN 2 THEN 'Fév' WHEN 3 THEN 'Mar'
                WHEN 4 THEN 'Avr' WHEN 5 THEN 'Mai' WHEN 6 THEN 'Juin'
                WHEN 7 THEN 'Juil' WHEN 8 THEN 'Août' WHEN 9 THEN 'Sept'
                WHEN 10 THEN 'Oct' WHEN 11 THEN 'Nov' WHEN 12 THEN 'Déc'
            END || ' ' || SUBSTRING(EXTRACT(YEAR FROM m.date_debut_n)::text FROM 3 FOR 2)
    END AS "Mois",
    m.i AS ordre,
    -- 12 derniers mois réels
    ROUND(COALESCE(SUM(f."TOTAL_HT") FILTER (
        WHERE f."Date_Facture" >= m.date_debut_n 
          AND f."Date_Facture" < m.date_fin_n
    ), 0)::numeric, 2) AS "CA_Recent",
    -- Même mois l'année précédente (N-1)
    ROUND(COALESCE(SUM(f."TOTAL_HT") FILTER (
        WHERE f."Date_Facture" >= m.date_debut_n1 
          AND f."Date_Facture" < m.date_fin_n1
    ), 0)::numeric, 2) AS "CA_Precedent"
FROM mois_glissants m
LEFT JOIN factures f ON (
    (f."Date_Facture" >= m.date_debut_n AND f."Date_Facture" < m.date_fin_n)
    OR 
    (f."Date_Facture" >= m.date_debut_n1 AND f."Date_Facture" < m.date_fin_n1)
)
GROUP BY m.i, m.date_debut_n, m.date_fin_n, m.date_debut_n1, m.date_fin_n1
ORDER BY m.i DESC;
